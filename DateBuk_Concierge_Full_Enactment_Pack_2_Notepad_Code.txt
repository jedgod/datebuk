import NextAuth from "next-auth";
import Credentials from "next-auth/providers/credentials";
import Google from "next-auth/providers/google";
import MicrosoftEntraID from "next-auth/providers/microsoft-entra-id";
import { PrismaAdapter } from "@auth/prisma-adapter";
import { compare } from "bcryptjs";
import { prisma } from "@/lib/prisma";
import { z } from "zod";

const credentialsSchema = z.object({
  email: z.string().email(),
  password: z.string().min(8),
});

export const { handlers, auth, signIn, signOut } = NextAuth({
  adapter: PrismaAdapter(prisma),
  session: {
    strategy: "jwt",
    maxAge: 30 * 24 * 60 * 60,
  },
  providers: [
    Credentials({
      credentials: {
        email: { label: "Email", type: "email" },
        password: { label: "Password", type: "password" },
      },
      authorize: async (rawCredentials) => {
        const credentials = credentialsSchema.parse(rawCredentials);

        const user = await prisma.user.findUnique({
          where: { email: credentials.email.toLowerCase() },
          select: {
            id: true,
            email: true,
            name: true,
            image: true,
            passwordHash: true,
            role: true,
            membershipTier: true,
            isSuspended: true,
          },
        });

        if (!user || !user.passwordHash || user.isSuspended) {
          return null;
        }

        const valid = await compare(credentials.password, user.passwordHash);
        if (!valid) return null;

        return {
          id: user.id,
          email: user.email,
          name: user.name,
          image: user.image,
          role: user.role,
          membershipTier: user.membershipTier,
        };
      },
    }),

    ...(process.env.GOOGLE_CLIENT_ID && process.env.GOOGLE_CLIENT_SECRET
      ? [
          Google({
            clientId: process.env.GOOGLE_CLIENT_ID,
            clientSecret: process.env.GOOGLE_CLIENT_SECRET,
          }),
        ]
      : []),

    ...(process.env.MICROSOFT_ENTRA_ID_CLIENT_ID &&
    process.env.MICROSOFT_ENTRA_ID_CLIENT_SECRET &&
    process.env.MICROSOFT_ENTRA_ID_TENANT_ID
      ? [
          MicrosoftEntraID({
            clientId: process.env.MICROSOFT_ENTRA_ID_CLIENT_ID,
            clientSecret: process.env.MICROSOFT_ENTRA_ID_CLIENT_SECRET,
            issuer:
              "https://login.microsoftonline.com/" +
              process.env.MICROSOFT_ENTRA_ID_TENANT_ID +
              "/v2.0",
          }),
        ]
      : []),
  ],
  callbacks: {
    async jwt({ token, user }) {
      if (user?.id) {
        token.sub = user.id;
      }

      if (token.sub) {
        const dbUser = await prisma.user.findUnique({
          where: { id: token.sub },
          select: {
            role: true,
            membershipTier: true,
            isSuspended: true,
          },
        });

        token.role = dbUser?.role ?? "USER";
        token.membershipTier = dbUser?.membershipTier ?? "FREE";
        token.isSuspended = dbUser?.isSuspended ?? true;
      }

      return token;
    },

    async session({ session, token }) {
      if (session.user && token.sub) {
        session.user.id = token.sub;
        session.user.role = String(token.role);
        session.user.membershipTier = String(token.membershipTier);
        session.user.isSuspended = Boolean(token.isSuspended);
      }

      return session;
    },

    async signIn({ user }) {
      if (!user.email) return false;

      const dbUser = await prisma.user.findUnique({
        where: { email: user.email },
        select: { isSuspended: true },
      });

      return !dbUser?.isSuspended;
    },
  },
  pages: {
    signIn: "/login",
    error: "/login",
  },
  trustHost: true,
});

export class AppError extends Error {
  constructor(
    public code: string,
    public status: number,
    message?: string,
    public details?: Record<string, unknown>
  ) {
    super(message ?? code);
    this.name = "AppError";
  }
}

export function toErrorResponse(error: unknown) {
  if (error instanceof AppError) {
    return {
      status: error.status,
      body: {
        error: error.code,
        message: error.message,
        details: error.details,
      },
    };
  }

  if (error instanceof Error) {
    return {
      status: 400,
      body: {
        error: error.message,
      },
    };
  }

  return {
    status: 500,
    body: {
      error: "UNKNOWN_ERROR",
    },
  };
}

import { headers } from "next/headers";

export async function getRequestContext() {
  const requestHeaders = await headers();

  const forwardedFor = requestHeaders.get("x-forwarded-for");
  const ipAddress = forwardedFor
    ? forwardedFor.split(",")[0].trim()
    : requestHeaders.get("x-real-ip") ?? undefined;

  return {
    ipAddress,
    userAgent: requestHeaders.get("user-agent") ?? undefined,
    requestId:
      requestHeaders.get("x-request-id") ??
      crypto.randomUUID(),
  };
}

import { redis } from "@/lib/redis";
import { AppError } from "@/lib/errors";

export async function withDistributedLock<T>(
  key: string,
  ttlMs: number,
  operation: () => Promise<T>
): Promise<T> {
  const token = crypto.randomUUID();
  const acquired = await redis.set(key, token, "PX", ttlMs, "NX");

  if (acquired !== "OK") {
    throw new AppError("RESOURCE_BUSY", 409);
  }

  try {
    return await operation();
  } finally {
    const current = await redis.get(key);
    if (current === token) {
      await redis.del(key);
    }
  }
}

import { redis } from "@/lib/redis";
import { AppError } from "@/lib/errors";

export async function enforceRateLimit(
  key: string,
  limit = 10,
  windowSeconds = 60
) {
  const redisKey = `ratelimit:${key}`;
  const count = await redis.incr(redisKey);

  if (count === 1) {
    await redis.expire(redisKey, windowSeconds);
  }

  if (count > limit) {
    const ttl = await redis.ttl(redisKey);

    throw new AppError(
      "RATE_LIMITED",
      429,
      "Too many requests. Please try again later.",
      { retryAfterSeconds: Math.max(ttl, 1) }
    );
  }
}

const envDefaults: Record<FeatureFlagKey, boolean> = {
  CONCIERGE_ENABLED:
    process.env.CONCIERGE_FEATURE_ENABLED === "true",
  CONCIERGE_PURCHASE_ENABLED: true,
  CONCIERGE_REQUEST_ENABLED: true,
  CONCIERGE_STAFF_OUTREACH_ENABLED: true,
  PROFILE_REVIEW_ENABLED: true,
  PROFILE_OPTIMIZATION_ENABLED: true,
  PHOTO_REVIEW_ENABLED: true,
  BACKGROUND_CHECK_ENABLED: false,
  DATE_PLANNING_ENABLED: true,
  RELATIONSHIP_COACHING_ENABLED: true,
  PRIORITY_SUPPORT_ENABLED: true,
};

export async function isFeatureEnabled(key: FeatureFlagKey) {
  const flag = await prisma.featureFlag.findUnique({
    where: { key },
    select: { enabled: true },
  });

  return flag?.enabled ?? envDefaults[key];
}

export async function requireFeature(key: FeatureFlagKey) {
  const enabled = await isFeatureEnabled(key);

  if (!enabled) {
    throw new Error("FEATURE_DISABLED");
  }
}

type ModerationResult = {
  allowed: boolean;
  score: number;
  decision: "ALLOW" | "REVIEW" | "REJECT";
  normalizedText: string;
  matchedRules: string[];
};

const rejectPatterns: Array<[string, RegExp]> = [
  ["OFF_PLATFORM_PHONE", /(?:\+?\d[\d\s\-().]{7,}\d)/i],
  ["OFF_PLATFORM_EMAIL", /\b[A-Z0-9._%+-]+@[A-Z0-9.-]+\.[A-Z]{2,}\b/i],
  ["SOCIAL_HANDLE", /(?:instagram|insta|snapchat|telegram|whatsapp|signal)\s*[:@]?\s*\w+/i],
  ["THREAT", /\b(?:hurt|kill|attack|find where you live|you will regret)\b/i],
  ["COERCION", /\b(?:you owe me|must reply|make her reply|force him|force her)\b/i],
  ["SEXUAL_PRESSURE", /\b(?:send nudes|sex now|sleep with me|hook up tonight)\b/i],
  ["PAYMENT_OFFER", /\b(?:pay you|cash for|money if you reply)\b/i],
];

const reviewPatterns: Array<[string, RegExp]> = [
  ["ANGER", /\b(?:angry|ignored me|disrespectful|ghosted)\b/i],
  ["REPEATED_CONTACT", /\b(?:again and again|many times|keeps ignoring)\b/i],
  ["ACTIVITY_SURVEILLANCE", /\b(?:online but not replying|saw her active|last seen)\b/i],
];

export function moderateConciergeText(text: string): ModerationResult {
  const normalizedText = text
    .normalize("NFKC")
    .replace(/\s+/g, " ")
    .trim();

  const matchedRules: string[] = [];
  let score = 0;

  for (const [name, pattern] of rejectPatterns) {
    if (pattern.test(normalizedText)) {
      matchedRules.push(name);
      score += 50;
    }
  }

  for (const [name, pattern] of reviewPatterns) {
    if (pattern.test(normalizedText)) {
      matchedRules.push(name);
      score += 15;
    }
  }

  if (normalizedText.length < 10) {
    matchedRules.push("TOO_SHORT");
    score += 20;
  }

  if (score >= 50) {
    return {
      allowed: false,
      score,
      decision: "REJECT",
      normalizedText,
      matchedRules,
    };
  }

  if (score >= 15) {
    return {
      allowed: true,
      score,
      decision: "REVIEW",
      normalizedText,
      matchedRules,
    };
  }

  return {
    allowed: true,
    score,
    decision: "ALLOW",
    normalizedText,
    matchedRules,
  };
}

import { prisma } from "@/lib/prisma";
import { getConciergeOperations } from "@/lib/operational-settings";

export async function assignBestAvailableStaff(
  conciergeRequestId: string,
  assignedById?: string
) {
  const operations = await getConciergeOperations();

  const staffMembers = await prisma.user.findMany({
    where: {
      role: { in: ["STAFF", "ADMIN"] },
      isSuspended: false,
    },
    select: {
      id: true,
      name: true,
      _count: {
        select: {
          conciergeAssignedCases: {
            where: {
              status: { in: ["ASSIGNED", "ACCEPTED"] },
            },
          },
        },
      },
    },
  });

  const candidate = staffMembers
    .filter(
      (staff) =>
        staff._count.conciergeAssignedCases <
        operations.maxStaffActiveCases
    )
    .sort(
      (a, b) =>
        a._count.conciergeAssignedCases -
        b._count.conciergeAssignedCases
    )[0];

  if (!candidate) {
    return null;
  }

  return prisma.staffAssignment.upsert({
    where: { conciergeRequestId },
    update: {
      staffId: candidate.id,
      assignedById,
      status: "ASSIGNED",
      assignedAt: new Date(),
      releasedAt: null,
    },
    create: {
      conciergeRequestId,
      staffId: candidate.id,
      assignedById,
    },
  });
}

import { Resend } from "resend";
import { prisma } from "@/lib/prisma";

const resend = new Resend(process.env.RESEND_API_KEY);

export async function deliverEmail(input: {
  deliveryId: string;
  to: string;
  subject: string;
  html: string;
}) {
  const delivery = await prisma.notificationDelivery.findUnique({
    where: { id: input.deliveryId },
  });

  if (!delivery || delivery.status === "DELIVERED") {
    return;
  }

  try {
    const response = await resend.emails.send({
      from: process.env.EMAIL_FROM!,
      to: input.to,
      replyTo: process.env.EMAIL_REPLY_TO,
      subject: input.subject,
      html: input.html,
    });

    await prisma.notificationDelivery.update({
      where: { id: input.deliveryId },
      data: {
        provider: "resend",
        providerMessageId: response.data?.id,
        status: "SENT",
        attemptCount: { increment: 1 },
        sentAt: new Date(),
        lastError: null,
      },
    });
  } catch (error) {
    await prisma.notificationDelivery.update({
      where: { id: input.deliveryId },
      data: {
        status: "FAILED",
        attemptCount: { increment: 1 },
        lastError:
          error instanceof Error ? error.message : "EMAIL_SEND_FAILED",
        nextAttemptAt: new Date(Date.now() + 15 * 60 * 1000),
      },
    });

    throw error;
  }
}

import { NextResponse } from "next/server";
import { requireUser } from "@/lib/auth";
import { prisma } from "@/lib/prisma";
import { z } from "zod";
import { moderateConciergeText } from "@/lib/moderation";

const messageSchema = z.object({
  content: z.string().trim().min(1).max(500),
});

export async function GET(
  _request: Request,
  context: { params: Promise<{ id: string }> }
) {
  try {
    const user = await requireUser();
    const { id } = await context.params;

    const concierge = await prisma.conciergeRequest.findFirst({
      where: {
        id,
        OR: [
          { requesterId: user.id },
          { recipientId: user.id },
          ...(user.role !== "USER" ? [{}] : []),
        ],
      },
    });

    if (!concierge) {
      return NextResponse.json({ error: "NOT_FOUND" }, { status: 404 });
    }

    const messages = await prisma.conciergeMessage.findMany({
      where: {
        requestId: id,
        ...(user.role === "USER" ? { internal: false } : {}),
      },
      orderBy: { createdAt: "asc" },
    });

    return NextResponse.json(messages);
  } catch {
    return NextResponse.json({ error: "UNAUTHORIZED" }, { status: 401 });
  }
}

export async function POST(
  request: Request,
  context: { params: Promise<{ id: string }> }
) {
  try {
    const user = await requireUser();
    const { id } = await context.params;
    const body = messageSchema.parse(await request.json());

    const concierge = await prisma.conciergeRequest.findFirst({
      where: {
        id,
        status: "RECIPIENT_INTERESTED",
        OR: [
          { requesterId: user.id },
          { recipientId: user.id },
        ],
      },
    });

    if (!concierge) {
      return NextResponse.json(
        { error: "MESSAGING_NOT_AVAILABLE" },
        { status: 409 }
      );
    }

    const moderation = moderateConciergeText(body.content);

    if (!moderation.allowed) {
      return NextResponse.json(
        { error: "MESSAGE_REJECTED" },
        { status: 422 }
      );
    }

    const created = await prisma.conciergeMessage.create({
      data: {
        requestId: id,
        senderId: user.id,
        senderRole: user.role,
        content: moderation.normalizedText,
        internal: false,
      },
    });

    return NextResponse.json(created, { status: 201 });
  } catch (error) {
    const message = error instanceof Error ? error.message : "UNKNOWN_ERROR";
    return NextResponse.json({ error: message }, { status: 400 });
  }
}

import { prisma } from "@/lib/prisma";
import { stripe } from "@/lib/stripe";

export async function syncServiceCatalogToStripe() {
  const services = await prisma.serviceCatalog.findMany({
    where: { active: true },
    include: { stripeLink: true },
  });

  const results = [];

  for (const service of services) {
    let productId = service.stripeLink?.stripeProductId;

    if (!productId) {
      const product = await stripe.products.create({
        name: service.name,
        description: service.shortDescription,
        metadata: {
          serviceCatalogId: service.id,
          serviceType: service.type,
        },
      });

      productId = product.id;
    } else {
      await stripe.products.update(productId, {
        name: service.name,
        description: service.shortDescription,
        active: service.active,
      });
    }

    let priceId = service.stripeLink?.stripePriceId;
    let needsNewPrice = true;

    if (priceId) {
      const currentPrice = await stripe.prices.retrieve(priceId);
      needsNewPrice =
        currentPrice.unit_amount !== service.priceCents ||
        currentPrice.currency !== service.currency;
    }

    if (needsNewPrice) {
      const price = await stripe.prices.create({
        product: productId,
        unit_amount: service.priceCents,
        currency: service.currency,
        metadata: {
          serviceCatalogId: service.id,
        },
      });

      priceId = price.id;
    }

    const link = await prisma.stripeCatalogLink.upsert({
      where: { serviceCatalogId: service.id },
      update: {
        stripeProductId: productId,
        stripePriceId: priceId!,
        active: service.active,
        syncedAt: new Date(),
      },
      create: {
        serviceCatalogId: service.id,
        stripeProductId: productId,
        stripePriceId: priceId!,
        active: service.active,
      },
    });

    results.push(link);
  }

  return results;
}

import { NextResponse } from "next/server";
import { requireRole } from "@/lib/auth";
import { prisma } from "@/lib/prisma";

function csvEscape(value: unknown) {
  const stringValue =
    value === null || value === undefined
      ? ""
      : typeof value === "string"
      ? value
      : JSON.stringify(value);

  return `"${stringValue.replaceAll('"', '""')}"`;
}

export async function GET(request: Request) {
  try {
    await requireRole(["ADMIN"]);

    const url = new URL(request.url);
    const entityType = url.searchParams.get("entityType") ?? undefined;
    const from = url.searchParams.get("from");
    const to = url.searchParams.get("to");

    const logs = await prisma.auditLog.findMany({
      where: {
        ...(entityType ? { entityType } : {}),
        ...(from || to
          ? {
              createdAt: {
                ...(from ? { gte: new Date(from) } : {}),
                ...(to ? { lte: new Date(to) } : {}),
              },
            }
          : {}),
      },
      orderBy: { createdAt: "desc" },
      take: 50_000,
    });

    const header = [
      "id",
      "createdAt",
      "actorId",
      "action",
      "entityType",
      "entityId",
      "metadata",
      "ipAddress",
      "userAgent",
    ].join(",");

    const rows = logs.map((log) =>
      [
        log.id,
        log.createdAt.toISOString(),
        log.actorId,
        log.action,
        log.entityType,
        log.entityId,
        log.metadata,
        log.ipAddress,
        log.userAgent,
      ]
        .map(csvEscape)
        .join(",")
    );

    const csv = [header, ...rows].join("\n");

    return new NextResponse(csv, {
      headers: {
        "Content-Type": "text/csv; charset=utf-8",
        "Content-Disposition":
          'attachment; filename="datebuk-concierge-audit.csv"',
      },
    });
  } catch {
    return NextResponse.json({ error: "FORBIDDEN" }, { status: 403 });
  }
}

  });

import { NextResponse } from "next/server";
import { prisma } from "@/lib/prisma";
import { notificationQueue } from "@/lib/queue";

export async function POST(request: Request) {
  if (
    request.headers.get("authorization") !==
    `Bearer ${process.env.CRON_SECRET}`
  ) {
    return NextResponse.json({ error: "FORBIDDEN" }, { status: 403 });
  }

  const failed = await prisma.notificationDelivery.findMany({
    where: {
      status: "FAILED",
      attemptCount: { lt: 5 },
      OR: [
        { nextAttemptAt: null },
        { nextAttemptAt: { lte: new Date() } },
      ],
    },
    select: {
      notificationId: true,
      userId: true,
    },
    take: 500,
  });

  let queued = 0;

  for (const delivery of failed) {
    if (!delivery.notificationId) continue;

    await notificationQueue.add(
      "deliver-notification",
      {
        notificationId: delivery.notificationId,
        userId: delivery.userId,
      },
      {
        attempts: 5,
        backoff: { type: "exponential", delay: 5000 },
      }
    );

    queued += 1;
  }

  return NextResponse.json({ queued });
}

  const pending = await prisma.payment.findMany({

    );

  return NextResponse.json({ queued: pending.length });
}

import { prisma } from "@/lib/prisma";

export async function POST(request: Request) {
  /*
  In production, verify the webhook signature using the provider's
  official webhook verification method before processing.
  */

  const event = await request.json();
  const providerMessageId = event?.data?.email_id;

  if (!providerMessageId) {
    return NextResponse.json({ received: true });
  }

  const statusMap: Record<string, string> = {
    "email.sent": "SENT",
    "email.delivered": "DELIVERED",
    "email.bounced": "BOUNCED",
    "email.complained": "COMPLAINED",
  };

  const mappedStatus = statusMap[event.type];

  if (mappedStatus) {
    await prisma.notificationDelivery.updateMany({
      where: { providerMessageId },
      data: {
        status: mappedStatus as never,
        deliveredAt:
          mappedStatus === "DELIVERED" ? new Date() : undefined,
      },
    });
  }

  return NextResponse.json({ received: true });
}

import { useEffect, useState } from "react";

export function ConciergeSettingsForm() {

  const [saved, setSaved] = useState("");
  const [loading, setLoading] = useState(true);

  useEffect(() => {
    fetch("/api/concierge/settings")
      .then((response) => response.json())
      .then((data) => {
        setOptOut(Boolean(data.conciergeOptOut));
      })
      .finally(() => setLoading(false));

    setSaved("");

      method: "PATCH",
      headers: { "Content-Type": "application/json" },
      body: JSON.stringify({ conciergeOptOut: optOut }),
    });

    const data = await response.json();
    setSaved(response.ok ? "Settings saved." : data.error);
  }

  if (loading) {
    return <p>Loading settings...</p>;
  }

  return (
    <section className="rounded-3xl border bg-white p-6 shadow-sm">
      <h2 className="text-xl font-bold">Concierge Privacy</h2>

      <label className="mt-5 flex items-start gap-3">
        <input
          type="checkbox"
          checked={optOut}
          onChange={(event) => setOptOut(event.target.checked)}
          className="mt-1"
        />
        <span>
          <strong>Do not allow concierge introductions to me.</strong>
          <span className="mt-1 block text-sm text-slate-600">
            Staff will automatically reject future concierge requests
            involving your profile.

        onClick={save}

  );

"use client";

import { FormEvent, useState } from "react";

export function ConciergeMessageComposer({
  requestId,
}: {
  requestId: string;
}) {
  const [content, setContent] = useState("");
  const [result, setResult] = useState("");

  async function submit(event: FormEvent) {
    event.preventDefault();
    setResult("");

    const response = await fetch(
      `/api/concierge/requests/${requestId}/messages`,
      {
        method: "POST",
        headers: { "Content-Type": "application/json" },
        body: JSON.stringify({ content }),
      }
    );

    const data = await response.json();

    if (!response.ok) {
      setResult(data.error ?? "Message failed.");
      return;
    }

    setContent("");
    setResult("Message sent.");
    window.location.reload();
  }

  return (
    <form onSubmit={submit} className="rounded-2xl border p-4">
      <textarea
        value={content}
        onChange={(event) => setContent(event.target.value)}
        maxLength={500}
        rows={4}
        placeholder="Write a respectful in-platform message."
        className="w-full rounded-xl border p-3"
      />

      <div className="mt-2 flex items-center justify-between">
        <span className="text-xs text-slate-500">
          {content.length}/500
        </span>

        <button
          type="submit"
          className="rounded-xl bg-pink-500 px-4 py-2 font-semibold text-white"
        >
          Send
        </button>
      </div>

      {result && <p className="mt-2 text-sm">{result}</p>}
    </form>
  );
}

import { describe, expect, it } from "vitest";

function selectLeastLoadedStaff(
  staff: Array<{ id: string; activeCases: number }>,
  maxCases: number
) {
  return staff
    .filter((item) => item.activeCases < maxCases)
    .sort((a, b) => a.activeCases - b.activeCases)[0] ?? null;
}

describe("staff assignment", () => {
  it("selects the least-loaded eligible staff member", () => {
    const result = selectLeastLoadedStaff(
      [
        { id: "staff-a", activeCases: 5 },
        { id: "staff-b", activeCases: 2 },
        { id: "staff-c", activeCases: 8 },
      ],
      10
    );

    expect(result?.id).toBe("staff-b");
  });

  it("returns null when all staff are at capacity", () => {
    const result = selectLeastLoadedStaff(
      [
        { id: "staff-a", activeCases: 10 },
        { id: "staff-b", activeCases: 12 },
      ],
      10
    );

    expect(result).toBeNull();
  });
});

REFUND RULES
====================================================================

RETURN ONE CREDIT WHEN:
- Staff rejects before outreach.
- User cancels while still pending review.

DO NOT RETURN A CREDIT WHEN:
- Recipient declines.
- Recipient ignores the request.
- Recipient selects maybe later.
- Recipient blocks after outreach.
- Recipient reports after outreach, unless staff policy determines the
  requester should never have passed review.

REFUND MONEY WHEN:
- Duplicate payment.
- Service package was never delivered due to platform fault.
- Legally required.
