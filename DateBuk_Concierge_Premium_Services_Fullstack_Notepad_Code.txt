
DATEBUK CONCIERGE & PREMIUM SERVICES MODULE
===========================================

A production-oriented full-stack module for:
- Match Concierge outreach: $20 for 2 requests
- Profile review
- Profile optimization
- Photo review
- Background-check request intake
- Date-planning assistance
- Relationship-coaching booking
- Priority support
- User, Staff, and Admin dashboards
- Stripe checkout and webhook handling
- PostgreSQL + Prisma
- Consent, blocking, reporting, quotas, audit logs, and anti-harassment safeguards

STACK
-----
- Next.js 15+ App Router
- TypeScript
- PostgreSQL
- Prisma ORM
- NextAuth/Auth.js-compatible session hooks
- Stripe
- Zod validation
- Tailwind CSS

IMPORTANT
---------
This pack intentionally excludes live Stripe secrets, database credentials,
email/SMS provider credentials, and identity-verification vendor credentials.
Provide them through environment variables.

import { auth } from "@/auth";
import { prisma } from "@/lib/prisma";
import { UserRole } from "@prisma/client";

export async function requireUser() {
  const session = await auth();

  if (!session?.user?.email) {
    throw new Error("UNAUTHORIZED");
  }

  const user = await prisma.user.findUnique({
    where: { email: session.user.email },
  });

  if (!user || user.isSuspended) {
    throw new Error("UNAUTHORIZED");
  }

  return user;
}

export async function requireRole(roles: UserRole[]) {
  const user = await requireUser();

  if (!roles.includes(user.role)) {
    throw new Error("FORBIDDEN");
  }

  return user;
}

import { prisma } from "@/lib/prisma";
import { MembershipTier } from "@prisma/client";

const cooldownDays = Number(process.env.CONCIERGE_COOLDOWN_DAYS ?? 30);
const expiryDays = Number(process.env.CONCIERGE_REQUEST_EXPIRY_DAYS ?? 7);

export function conciergeExpiryDate() {
  const date = new Date();
  date.setDate(date.getDate() + expiryDays);
  return date;
}

export function monthlyLimitForTier(tier: MembershipTier) {
  if (tier === "ELITE") {
    return Number(process.env.CONCIERGE_MONTHLY_LIMIT_ELITE ?? 10);
  }

  if (tier === "PREMIUM") {
    return Number(process.env.CONCIERGE_MONTHLY_LIMIT_PREMIUM ?? 20);
  }

  return 0;
}

export async function evaluateConciergeEligibility(
  requesterId: string,
  recipientId: string
) {
  if (requesterId === recipientId) {
    return { allowed: false, code: "SELF_REQUEST" };
  }

  const [requester, recipient, wallet, directBlock, reverseBlock, unresolvedReport] =
    await Promise.all([
      prisma.user.findUnique({ where: { id: requesterId } }),
      prisma.user.findUnique({ where: { id: recipientId } }),
      prisma.conciergeWallet.findUnique({ where: { userId: requesterId } }),
      prisma.userBlock.findUnique({
        where: { blockerId_blockedId: { blockerId: requesterId, blockedId: recipientId } },
      }),
      prisma.userBlock.findUnique({
        where: { blockerId_blockedId: { blockerId: recipientId, blockedId: requesterId } },
      }),
      prisma.safetyReport.findFirst({
        where: {
          resolved: false,
          OR: [
            { reporterId: requesterId, reportedId: recipientId },
            { reporterId: recipientId, reportedId: requesterId },
          ],
        },
      }),
    ]);

  if (!requester || !recipient) return { allowed: false, code: "USER_NOT_FOUND" };
  if (requester.isSuspended || recipient.isSuspended) {
    return { allowed: false, code: "SUSPENDED_ACCOUNT" };
  }

  if (requester.membershipTier === "FREE") {
    return { allowed: false, code: "UPGRADE_REQUIRED" };
  }

  if (recipient.conciergeOptOut) {
    return { allowed: false, code: "RECIPIENT_OPTED_OUT" };
  }

  if (directBlock || reverseBlock) {
    return { allowed: false, code: "BLOCKED_RELATIONSHIP" };
  }

  if (unresolvedReport) {
    return { allowed: false, code: "SAFETY_REVIEW_REQUIRED" };
  }

  if (!wallet || wallet.availableCredits < 1) {
    return { allowed: false, code: "NO_CREDITS" };
  }

  const cooldownStart = new Date();
  cooldownStart.setDate(cooldownStart.getDate() - cooldownDays);

  const recentPairRequest = await prisma.conciergeRequest.findFirst({
    where: {
      requesterId,
      recipientId,
      createdAt: { gte: cooldownStart },
      status: {
        notIn: ["REJECTED", "CANCELLED"],
      },
    },
  });

  if (recentPairRequest) {
    return { allowed: false, code: "PAIR_COOLDOWN_ACTIVE" };
  }

  const monthStart = new Date();
  monthStart.setDate(1);
  monthStart.setHours(0, 0, 0, 0);

  const monthlyCount = await prisma.conciergeRequest.count({
    where: {
      requesterId,
      createdAt: { gte: monthStart },
      status: { not: "CANCELLED" },
    },
  });

  const limit = monthlyLimitForTier(requester.membershipTier);
  if (monthlyCount >= limit) {
    return { allowed: false, code: "MONTHLY_LIMIT_REACHED" };
  }

  return {
    allowed: true,
    code: "ELIGIBLE",
    walletCredits: wallet.availableCredits,
    monthlyUsed: monthlyCount,
    monthlyLimit: limit,
  };
}

const buckets = new Map<string, { count: number; resetAt: number }>();

export function enforceRateLimit(
  key: string,
  limit = 10,
  windowMs = 60_000
) {
  const now = Date.now();
  const current = buckets.get(key);

  if (!current || current.resetAt <= now) {
    buckets.set(key, { count: 1, resetAt: now + windowMs });
    return;
  }

  if (current.count >= limit) {
    throw new Error("RATE_LIMITED");
  }

  current.count += 1;
}

import { NextResponse } from "next/server";
import { requireUser } from "@/lib/auth";
import { evaluateConciergeEligibility } from "@/lib/concierge-policy";

export async function GET(
  _request: Request,
  context: { params: Promise<{ recipientId: string }> }
) {
  try {
    const user = await requireUser();
    const { recipientId } = await context.params;

    const result = await evaluateConciergeEligibility(user.id, recipientId);
    return NextResponse.json(result);
  } catch (error) {
    const message = error instanceof Error ? error.message : "UNKNOWN_ERROR";
    return NextResponse.json({ error: message }, { status: 401 });
  }
}

import { NextResponse } from "next/server";
import { requireUser } from "@/lib/auth";
import { prisma } from "@/lib/prisma";
import { createConciergeRequestSchema } from "@/lib/validators";
import {
  conciergeExpiryDate,
  evaluateConciergeEligibility,
} from "@/lib/concierge-policy";
import { writeAuditLog } from "@/lib/audit";
import { createNotification } from "@/lib/notifications";
import { enforceRateLimit } from "@/lib/rate-limit";

export async function GET() {
  try {
    const user = await requireUser();

    const requests = await prisma.conciergeRequest.findMany({
      where: {
        OR: [{ requesterId: user.id }, { recipientId: user.id }],
      },
      include: {
        requester: { select: { id: true, name: true, image: true } },
        recipient: { select: { id: true, name: true, image: true } },
      },
      orderBy: { createdAt: "desc" },
    });

    return NextResponse.json(requests);
  } catch {
    return NextResponse.json({ error: "UNAUTHORIZED" }, { status: 401 });
  }

export async function POST(request: Request) {
  try {
    const user = await requireUser();
    enforceRateLimit(`concierge-create:${user.id}`, 5, 60_000);

    const body = createConciergeRequestSchema.parse(await request.json());
    const eligibility = await evaluateConciergeEligibility(
      user.id,
      body.recipientId
    );

    if (!eligibility.allowed) {
      return NextResponse.json(
        { error: eligibility.code },
        { status: 409 }
      );
    }

    const created = await prisma.$transaction(async (tx) => {
      const wallet = await tx.conciergeWallet.updateMany({
        where: {
          userId: user.id,
          availableCredits: { gte: 1 },
        },
        data: {
          availableCredits: { decrement: 1 },
          lifetimeUsed: { increment: 1 },
        },
      });

      if (wallet.count !== 1) {
        throw new Error("NO_CREDITS");
      }

      return tx.conciergeRequest.create({
        data: {
          requesterId: user.id,
          recipientId: body.recipientId,
          reason: body.reason,
          requesterMessage: body.requesterMessage,
          creditConsumed: true,
          expiresAt: conciergeExpiryDate(),
          messages: {
            create: {
              senderId: user.id,
              senderRole: user.role,
              content: body.requesterMessage,
              internal: true,
            },
          },
        },
      });
    });

    await writeAuditLog({
      actorId: user.id,
      action: "CONCIERGE_REQUEST_CREATED",
      entityType: "ConciergeRequest",
      entityId: created.id,
      conciergeRequestId: created.id,
      metadata: { recipientId: body.recipientId, reason: body.reason },
    });

    await createNotification({
      userId: user.id,
      type: "CONCIERGE_SUBMITTED",
      title: "Concierge request submitted",
      body: "Your request is waiting for staff review.",
      data: { requestId: created.id },
    });

    return NextResponse.json(created, { status: 201 });
  } catch (error) {
    const message = error instanceof Error ? error.message : "UNKNOWN_ERROR";

    const status =
      message === "UNAUTHORIZED"
        ? 401
        : message === "RATE_LIMITED"
        ? 429
        : message === "NO_CREDITS"
        ? 409
        : 400;

    return NextResponse.json({ error: message }, { status });
  }
}

import { NextResponse } from "next/server";
import { requireUser } from "@/lib/auth";
import { prisma } from "@/lib/prisma";
import { recipientResponseSchema } from "@/lib/validators";
import { createNotification } from "@/lib/notifications";
import { writeAuditLog } from "@/lib/audit";

export async function POST(
  request: Request,
  context: { params: Promise<{ id: string }> }
) {
  try {
    const user = await requireUser();
    const { id } = await context.params;
    const body = recipientResponseSchema.parse(await request.json());

    const concierge = await prisma.conciergeRequest.findFirst({
      where: {
        id,
        recipientId: user.id,
        status: "OUTREACH_SENT",
      },
    });

    if (!concierge) {
      return NextResponse.json(
        { error: "REQUEST_NOT_AVAILABLE" },
        { status: 404 }
      );
    }

    const statusMap = {
      INTERESTED: "RECIPIENT_INTERESTED",
      REPLY_NOW: "RECIPIENT_INTERESTED",
      MAYBE_LATER: "OUTREACH_SENT",
      NOT_INTERESTED: "RECIPIENT_DECLINED",
      BLOCK_REQUESTER: "RECIPIENT_DECLINED",
      OPT_OUT: "RECIPIENT_DECLINED",
      REPORT: "RECIPIENT_DECLINED",
    } as const;

    const result = await prisma.$transaction(async (tx) => {
      if (body.response === "BLOCK_REQUESTER") {
        await tx.userBlock.upsert({
          where: {
            blockerId_blockedId: {
              blockerId: user.id,
              blockedId: concierge.requesterId,
            },
          },
          update: {},
          create: {
            blockerId: user.id,
            blockedId: concierge.requesterId,
          },
        });
      }

      if (body.response === "OPT_OUT") {
        await tx.user.update({
          where: { id: user.id },
          data: { conciergeOptOut: true },
        });
      }

      if (body.response === "REPORT") {
        await tx.safetyReport.create({
          data: {
            reporterId: user.id,
            reportedId: concierge.requesterId,
            category: "CONCIERGE_OUTREACH_REPORT",
            description: body.message,
          },
        });
      }

      return tx.conciergeRequest.update({
        where: { id },
        data: {
          recipientResponse: body.response,
          recipientMessage: body.message,
          status: statusMap[body.response],
          respondedAt: new Date(),
          completedAt:
            body.response === "MAYBE_LATER" ? undefined : new Date(),
          messages: body.message
            ? {
                create: {
                  senderId: user.id,
                  senderRole: user.role,
                  content: body.message,
                  internal: false,
                },
              }
            : undefined,
        },
      });
    });

    const requesterBody =
      body.response === "INTERESTED" || body.response === "REPLY_NOW"
        ? "The member is open to continuing the conversation."
        : body.response === "MAYBE_LATER"
        ? "The member selected maybe later."
        : "The concierge request has been closed.";

    await createNotification({
      userId: concierge.requesterId,
      type: "CONCIERGE_RESPONSE_RECEIVED",
      title: "Concierge request updated",
      body: requesterBody,
      data: { requestId: id },
    });

    await writeAuditLog({
      actorId: user.id,
      action: "CONCIERGE_RECIPIENT_RESPONSE",
      entityType: "ConciergeRequest",
      entityId: id,
      conciergeRequestId: id,
      metadata: { response: body.response },
    });

    return NextResponse.json(result);
  } catch (error) {
    const message = error instanceof Error ? error.message : "UNKNOWN_ERROR";
    return NextResponse.json({ error: message }, { status: 400 });
  }
}

export async function POST(
  request: Request,

    const staff = await requireRole(["STAFF", "ADMIN"]);
    const { id } = await context.params;
    const body = staffActionSchema.parse(await request.json());

    const concierge = await prisma.conciergeRequest.findUnique({
      where: { id },
      include: {
        requester: true,
        recipient: true,
      },
    });

    if (!concierge) {
      return NextResponse.json({ error: "NOT_FOUND" }, { status: 404 });
    }

    if (body.action === "APPROVE") {
      const updated = await prisma.conciergeRequest.update({
        where: { id },
        data: {
          status: "APPROVED",
          staffId: staff.id,
          staffNotes: body.staffNotes,
        },
      });

      await createNotification({
        userId: concierge.requesterId,
        type: "CONCIERGE_APPROVED",
        title: "Concierge request approved",
        body: "A staff member has approved your request.",
        data: { requestId: id },
      });

      return NextResponse.json(updated);
    }

    if (body.action === "REJECT") {
      const updated = await prisma.$transaction(async (tx) => {
        if (concierge.creditConsumed) {
          await tx.conciergeWallet.update({
            where: { userId: concierge.requesterId },
            data: {
              availableCredits: { increment: 1 },
              lifetimeUsed: { decrement: 1 },
            },
          });
        }

        return tx.conciergeRequest.update({
          where: { id },
          data: {
            status: "REJECTED",
            staffId: staff.id,
            staffNotes: body.staffNotes,
            creditConsumed: false,
            completedAt: new Date(),
          },
        });
      });

      await createNotification({
        userId: concierge.requesterId,
        type: "CONCIERGE_REJECTED",
        title: "Concierge request not approved",
        body: "The request was closed and its credit was returned.",
        data: { requestId: id },
      });

      return NextResponse.json(updated);
    }

    if (body.action === "SEND_OUTREACH") {
      if (concierge.status !== "APPROVED") {
        return NextResponse.json(
          { error: "REQUEST_MUST_BE_APPROVED" },
          { status: 409 }
        );
      }

      const neutralMessage =
        body.staffMessage ??
        "A member has requested a one-time concierge introduction. " +
          "There is no obligation to respond. You may accept, decline, block, " +
          "report, or opt out of future concierge requests.";

      const updated = await prisma.conciergeRequest.update({
        where: { id },
        data: {

          staffId: staff.id,
          staffMessage: neutralMessage,
          staffNotes: body.staffNotes,
          outreachSentAt: new Date(),
          messages: {
            create: {
              senderId: staff.id,
              senderRole: staff.role,
              content: neutralMessage,
              internal: false,
            },
          },
        },
      });

      await createNotification({
        userId: concierge.recipientId,
        type: "CONCIERGE_OUTREACH_SENT",
        title: "You received a concierge introduction",

      });

        userId: concierge.requesterId,
        type: "CONCIERGE_OUTREACH_SENT",
        title: "Concierge outreach sent",
        body: "The recipient can now respond privately through the platform.",
        data: { requestId: id },
      });

      return NextResponse.json(updated);
    }

    if (body.action === "REFUND_CREDIT") {
      const updated = await prisma.$transaction(async (tx) => {
        if (concierge.creditConsumed) {
          await tx.conciergeWallet.update({
            where: { userId: concierge.requesterId },
            data: {
              availableCredits: { increment: 1 },
              lifetimeUsed: { decrement: 1 },
            },
          });
        }

        return tx.conciergeRequest.update({
          where: { id },
          data: { creditConsumed: false },
        });
      });

      return NextResponse.json(updated);
    }

    const updated = await prisma.conciergeRequest.update({
      where: { id },
      data: {
        status: "CLOSED",
        completedAt: new Date(),
        staffId: staff.id,
        staffNotes: body.staffNotes,
      },
    });

    await writeAuditLog({
      actorId: staff.id,
      action: `CONCIERGE_STAFF_${body.action}`,
      entityType: "ConciergeRequest",
      entityId: id,
      conciergeRequestId: id,
      metadata: {
        staffNotes: body.staffNotes,
      },
    });

    return NextResponse.json(updated);
  } catch (error) {
    const message = error instanceof Error ? error.message : "UNKNOWN_ERROR";
    return NextResponse.json({ error: message }, { status: 400 });
  }
}

import { NextResponse } from "next/server";
import Stripe from "stripe";
import { stripe } from "@/lib/stripe";
import { prisma } from "@/lib/prisma";
import { createNotification } from "@/lib/notifications";
import { writeAuditLog } from "@/lib/audit";

export const runtime = "nodejs";

export async function POST(request: Request) {
  const signature = request.headers.get("stripe-signature");

  if (!signature) {
    return NextResponse.json({ error: "MISSING_SIGNATURE" }, { status: 400 });
  }

  const rawBody = await request.text();

  let event: Stripe.Event;

  try {
    event = stripe.webhooks.constructEvent(
      rawBody,
      signature,
      process.env.STRIPE_WEBHOOK_SECRET!
    );
  } catch {
    return NextResponse.json({ error: "INVALID_SIGNATURE" }, { status: 400 });
  }

  if (event.type === "checkout.session.completed") {
    const session = event.data.object;
    const paymentId = session.metadata?.paymentId;
    const serviceOrderId = session.metadata?.serviceOrderId;
    const userId = session.metadata?.userId;
    const serviceType = session.metadata?.serviceType;

    if (paymentId && serviceOrderId && userId) {
      await prisma.$transaction(async (tx) => {
        const existing = await tx.payment.findUnique({
          where: { id: paymentId },
        });

        if (!existing || existing.status === "PAID") {
          return;
        }

        await tx.payment.update({
          where: { id: paymentId },
          data: {
            status: "PAID",
            stripePaymentIntentId:
              typeof session.payment_intent === "string"
                ? session.payment_intent
                : undefined,
          },
        });

        await tx.serviceOrder.update({
          where: { id: serviceOrderId },
          data: { status: "PAID" },
        });

        if (serviceType === "CONCIERGE_OUTREACH_PACKAGE") {
          const units = Number(
            (existing.metadata as { includedUnits?: number } | null)
              ?.includedUnits ?? 2
          );

          await tx.conciergeWallet.upsert({
            where: { userId },
            update: {
              availableCredits: { increment: units },
              lifetimePurchased: { increment: units },
            },
            create: {
              userId,
              availableCredits: units,
              lifetimePurchased: units,
            },
          });

          await tx.serviceOrder.update({
            where: { id: serviceOrderId },
            data: {
              status: "COMPLETED",
              completedAt: new Date(),
              deliveryJson: {
                creditsAdded: units,
              },
            },
          });
        }
      });

      await createNotification({
        userId,
        type: "PAYMENT_RECEIPT",
        title: "Payment received",
        body:
          serviceType === "CONCIERGE_OUTREACH_PACKAGE"
            ? "Your concierge credits are now available."
            : "Your premium service order has been received.",
        data: { serviceOrderId },
      });

      await writeAuditLog({
        actorId: userId,
        action: "PAYMENT_COMPLETED",
        entityType: "ServiceOrder",
        entityId: serviceOrderId,
        serviceOrderId,
        metadata: { stripeSessionId: session.id },
      });
    }
  }

  if (event.type === "checkout.session.expired") {
    const session = event.data.object;
    const paymentId = session.metadata?.paymentId;

    if (paymentId) {
      await prisma.payment.updateMany({
        where: { id: paymentId, status: "PENDING" },
        data: { status: "FAILED" },
      });
    }
  }

  return NextResponse.json({ received: true });
}

export function middleware(request: NextRequest) {
  const pathname = request.nextUrl.pathname;

}

PRODUCTION SAFETY RULES
====================================================================

1. Never contact someone who blocked or reported the requester.
2. Never reveal last-seen status, message-read status, location, phone number,
   email address, social-media account, or staff-only risk data.
3. Never imply the recipient is interested before explicit consent.
4. Never permit repeated outreach inside the pair cooldown window.
5. Never let staff manually bypass a recipient's opt-out.
6. Return the credit when staff rejects a request before outreach.
7. Do not return the credit merely because the recipient declines.
8. Disable concierge access for suspended or abusive accounts.
9. Preserve immutable audit logs for every payment, approval, rejection,
   outreach, response, block, report, and refund.
10. Use a consent-based approved vendor for background checks.
11. Do not describe coaching as therapy, treatment, or crisis support.
12. Add staff training, quality review, and escalation procedures.
13. Add automatic expiry for unanswered requests.
14. Rate-limit creation, responses, and staff actions.
15. Add fraud controls for payment disputes and repeated refunds.

import { NextResponse } from "next/server";
import { prisma } from "@/lib/prisma";

export async function POST(request: Request) {
  const authorization = request.headers.get("authorization");

  if (authorization !== `Bearer ${process.env.CRON_SECRET}`) {
    return NextResponse.json({ error: "FORBIDDEN" }, { status: 403 });
  }

  const result = await prisma.conciergeRequest.updateMany({
    where: {
      status: "OUTREACH_SENT",
      expiresAt: { lt: new Date() },
    },
    data: {
      status: "EXPIRED",
      completedAt: new Date(),
    },
  });

  return NextResponse.json({ expired: result.count });
}

CRON_SECRET="replace-with-a-long-random-secret"
